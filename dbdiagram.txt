Table users{
  id int [primary key]
  username nvarchar
  email varchar
  password varchar
  avatar varchar
  phone int
  create_at datetime
  update_at datetime
}
Table categories{
  id int [primary key]
  name varchar
  image text
}
Table products{
  id int [primary key]
  name varchar
  descriptions varchar
  price numeric(12,0)
  oldprice numeric(12,0)
  category_id int
  specification varchar
  image_url text
  byturn int // xem số lượng bao nhiêu lần sản phẩm được mua
  quantity int
  create_at datetime
  update_at datetime
}
ref: "categories"."id"< "products"."category_id" // một danh mục sản phẩm có thể có nhiều sản phẩm, 1 sản phẩm chỉ thuộc một danh mục - quan hệ (1,n)
Table Feedback{
  id int [primary key]
  product_id int
  user_id int
  start int
  context text
  create_at datetime
  update_at datetime
}

Ref: "Feedback"."user_id" > "users"."id"
